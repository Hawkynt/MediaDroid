# libass (ISC) and what it draws text with, as static libraries inside MediaDroid's own: FreeType (FTL, the licence
# taken of its FTL/GPLv2 choice), FriBidi (LGPL-2.1-or-later) and HarfBuzz (MIT). ASS_SOURCES holds each one's release
# unpacked in a folder of its name (native/build.gradle.kts, assSources, fetches and checks them); their notices are
# app/THIRD-PARTY-NOTICE.libass.txt.
#
# The sources are built as they come. Only FreeType has a CMake build its authors keep; for FriBidi, HarfBuzz and libass
# the files and settings below are those of their own Meson builds, without anything those make optional: no assembly
# (libass's C versions of its blur and rasterizer), no iconv (the subtitles reach libass as UTF-8), no libunibreak.
# The fonts come from the file (its attachments), a default font the app names, and the system's own where libass has a
# way to ask it: DirectWrite on Windows, Fontconfig on Linux where it is installed to build against. Android has none.

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)

# FreeType through its own build, without the libraries it can use and libass does not need.
set(BUILD_SHARED_LIBS OFF)
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON)
add_subdirectory("${ASS_SOURCES}/freetype" freetype EXCLUDE_FROM_ALL)

# FriBidi: its library's sources, its tables as its release carries them already generated.
set(FRIBIDI "${ASS_SOURCES}/fribidi")
set(FRIBIDI_VERSION 1.0.17)
set(FRIBIDI_INTERFACE_VERSION 4)
string(REPLACE "." ";" FRIBIDI_PARTS ${FRIBIDI_VERSION})
list(GET FRIBIDI_PARTS 0 FRIBIDI_MAJOR_VERSION)
list(GET FRIBIDI_PARTS 1 FRIBIDI_MINOR_VERSION)
list(GET FRIBIDI_PARTS 2 FRIBIDI_MICRO_VERSION)
set(PACKAGE fribidi)
set(PACKAGE_NAME "GNU FriBidi")
set(PACKAGE_BUGREPORT "https://github.com/fribidi/fribidi/issues/new")
set(SIZEOF_INT 4)
if(MSVC)
    set(FRIBIDI_MSVC_BUILD_PLACEHOLDER "#define FRIBIDI_BUILT_WITH_MSVC")
else()
    set(FRIBIDI_MSVC_BUILD_PLACEHOLDER "#undef FRIBIDI_BUILT_WITH_MSVC")
endif()
set(configure_input "fribidi-config.h generated by MediaDroid's native/ass/CMakeLists.txt")
configure_file("${FRIBIDI}/lib/fribidi-config.h.in" "${CMAKE_CURRENT_BINARY_DIR}/fribidi/fribidi-config.h" @ONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fribidi/config.h"
    "#define HAVE_MEMMOVE 1\n#define HAVE_MEMSET 1\n#define HAVE_STRDUP 1\n#define HAVE_STDLIB_H 1\n"
    "#define HAVE_STRING_H 1\n#define HAVE_STRINGIZE 1\n#define STDC_HEADERS 1\n#define FRIBIDI_NO_DEPRECATED 0\n")
add_library(ass_fribidi STATIC
    ${FRIBIDI}/lib/fribidi.c ${FRIBIDI}/lib/fribidi-arabic.c ${FRIBIDI}/lib/fribidi-bidi.c
    ${FRIBIDI}/lib/fribidi-bidi-types.c ${FRIBIDI}/lib/fribidi-char-sets.c ${FRIBIDI}/lib/fribidi-char-sets-cap-rtl.c
    ${FRIBIDI}/lib/fribidi-char-sets-cp1255.c ${FRIBIDI}/lib/fribidi-char-sets-cp1256.c
    ${FRIBIDI}/lib/fribidi-char-sets-iso8859-6.c ${FRIBIDI}/lib/fribidi-char-sets-iso8859-8.c
    ${FRIBIDI}/lib/fribidi-char-sets-utf8.c ${FRIBIDI}/lib/fribidi-deprecated.c ${FRIBIDI}/lib/fribidi-joining.c
    ${FRIBIDI}/lib/fribidi-joining-types.c ${FRIBIDI}/lib/fribidi-mirroring.c ${FRIBIDI}/lib/fribidi-brackets.c
    ${FRIBIDI}/lib/fribidi-run.c ${FRIBIDI}/lib/fribidi-shape.c)
target_include_directories(ass_fribidi PUBLIC "${FRIBIDI}/lib" "${CMAKE_CURRENT_BINARY_DIR}/fribidi")
target_compile_definitions(ass_fribidi PUBLIC FRIBIDI_LIB_STATIC PRIVATE HAVE_CONFIG_H)

# HarfBuzz: its single-file build, with nothing it can sit on top of (no GLib, ICU, FreeType or Uniscribe).
add_library(ass_harfbuzz STATIC "${ASS_SOURCES}/harfbuzz/src/harfbuzz.cc")
target_include_directories(ass_harfbuzz PUBLIC "${ASS_SOURCES}/harfbuzz/src")
set_target_properties(ass_harfbuzz PROPERTIES CXX_STANDARD 17)
if(MSVC)
    target_compile_options(ass_harfbuzz PRIVATE /bigobj)
else()
    target_compile_options(ass_harfbuzz PRIVATE -fno-exceptions -fno-rtti -fno-threadsafe-statics)
endif()

# libass.
set(LIBASS "${ASS_SOURCES}/libass/libass")
set(LIBASS_SOURCES
    ${LIBASS}/c/c_be_blur.c ${LIBASS}/c/c_blend_bitmaps.c ${LIBASS}/c/c_blur.c ${LIBASS}/c/c_rasterizer.c
    ${LIBASS}/ass.c ${LIBASS}/ass_arabic_charmap.c ${LIBASS}/ass_bitmap.c ${LIBASS}/ass_bitmap_engine.c
    ${LIBASS}/ass_blur.c ${LIBASS}/ass_cache.c ${LIBASS}/ass_drawing.c ${LIBASS}/ass_filesystem.c
    ${LIBASS}/ass_font.c ${LIBASS}/ass_fontselect.c ${LIBASS}/ass_library.c ${LIBASS}/ass_outline.c
    ${LIBASS}/ass_parse.c ${LIBASS}/ass_rasterizer.c ${LIBASS}/ass_render.c ${LIBASS}/ass_render_api.c
    ${LIBASS}/ass_shaper.c ${LIBASS}/ass_string.c ${LIBASS}/ass_strtod.c ${LIBASS}/ass_utils.c)
set(LIBASS_CONFIG "#define CONFIG_LARGE_TILES 0\n#define CONFIG_SOURCEVERSION \"release 0.17.5\"\n")
set(LIBASS_LINK)
if(WIN32)
    list(APPEND LIBASS_SOURCES ${LIBASS}/ass_directwrite.c)
    string(APPEND LIBASS_CONFIG "#define CONFIG_DIRECTWRITE 1\n")
    list(APPEND LIBASS_LINK gdi32)
elseif(NOT ANDROID)
    find_package(Fontconfig QUIET)
    if(Fontconfig_FOUND)
        list(APPEND LIBASS_SOURCES ${LIBASS}/ass_fontconfig.c)
        string(APPEND LIBASS_CONFIG "#define CONFIG_FONTCONFIG 1\n")
        list(APPEND LIBASS_LINK Fontconfig::Fontconfig)
    endif()
endif()
if(NOT MSVC)
    string(APPEND LIBASS_CONFIG "#define HAVE_STRDUP 1\n#define HAVE_STRNDUP 1\n")
endif()
if(NOT WIN32)
    string(APPEND LIBASS_CONFIG "#define HAVE_FSTAT 1\n")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/libass/config.h" "${LIBASS_CONFIG}")
add_library(ass STATIC ${LIBASS_SOURCES})
target_include_directories(ass PUBLIC "${LIBASS}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/libass")
target_compile_definitions(ass PRIVATE CONFIG_FREETYPE=1 CONFIG_FRIBIDI=1 CONFIG_HARFBUZZ=1)
set_target_properties(ass PROPERTIES C_STANDARD 11)
if(NOT MSVC)
    target_compile_definitions(ass PRIVATE _GNU_SOURCE)
    target_compile_options(ass PRIVATE -fno-math-errno)
endif()
target_link_libraries(ass PUBLIC freetype ass_fribidi ass_harfbuzz ${LIBASS_LINK})
if(NOT WIN32)
    find_library(MATH_LIBRARY m)
    if(MATH_LIBRARY)
        target_link_libraries(ass PUBLIC ${MATH_LIBRARY})
    endif()
endif()

# Small rather than fast, in every build: subtitles are a few glyphs a frame, and every app ABI carries the stack.
foreach(library freetype ass_fribidi ass_harfbuzz ass)
    if(MSVC)
        target_compile_options(${library} PRIVATE /O1)
    else()
        target_compile_options(${library} PRIVATE -Os)
    endif()
endforeach()

# MediaDroid's own: a track drawn as one picture (overlay.h), for jni/subtitles.cpp.
add_library(mediadroid_subtitles STATIC overlay.cpp)
target_include_directories(mediadroid_subtitles PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mediadroid_subtitles PUBLIC ass)
